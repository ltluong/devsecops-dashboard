import { useEffect, useState } from "react";

export default function GatesDashboard() {
  const [result, setResult] = useState(null);

  useEffect(() => {
    fetch("http://127.0.0.1:8001/pipeline-result")
      .then((res) => res.json())
      .then((data) => setResult(data))
      .catch((err) => console.error("API Error:", err));
  }, []);

  if (!result) {
    return (
      <div
        style={{
          background: "#071633",
          color: "white",
          minHeight: "100vh",
          padding: "30px",
        }}
      >
        Loading...
      </div>
    );
  }

  const gates = [
    { id: "1.1", name: "SonarQube", status: result.sonarqube },
    { id: "1.2", name: "Semgrep", status: result.semgrep },
    { id: "1.3", name: "Trivy FS", status: result.trivy_fs },
    { id: "1.4", name: "Coverage", status: result.coverage },
    { id: "2.1", name: "Trivy Image", status: result.trivy_image },
    { id: "2.2", name: "Checkov", status: result.checkov },
    { id: "3.1", name: "OWASP ZAP", status: result.zap },
    { id: "3.2", name: "Release Gate", status: result.release_decision },
  ];

  return (
    <div
      style={{
        background: "#071633",
        minHeight: "100vh",
        color: "white",
        padding: "30px",
      }}
    >
      <h1
        style={{
          textAlign: "center",
          marginBottom: "10px",
        }}
      >
        DevSecOps Dashboard
      </h1>

      <h2
        style={{
          textAlign: "center",
          marginBottom: "30px",
        }}
      >
        Security Quality Gates
      </h2>

      <div
        style={{
          display: "grid",
          gridTemplateColumns: "repeat(4, 1fr)",
          gap: "20px",
        }}
      >
        {gates.map((gate) => (
          <div
            key={gate.id}
            style={{
              background: "#10203d",
              borderRadius: "12px",
              padding: "20px",
              textAlign: "center",
              border: "1px solid #1f3b6d",
            }}
          >
            <h3>Gate {gate.id}</h3>

            <p>{gate.name}</p>

            <h2
              style={{
                color:
                  gate.status === "PASS" || gate.status === "GO"
                    ? "#00ff88"
                    : "#ff4444",
              }}
            >
              {gate.status}
            </h2>
          </div>
        ))}
      </div>
    </div>
  );
}
